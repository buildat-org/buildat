# Client command sequence for a visual check. Run from Build/:
#   bin/buildat -s localhost -w 1600x900 \
#           -c @../apps/sample_lighting_scene/check.txt
#
# -w fixes the window size for the run without touching the remembered one, so
# the images are the same size whatever the window was left at last time.
# See doc/client_commands.txt.
#
# The benchmark cameras are keys 1 to 6, so nothing here depends on timing a
# movement, and the images are directly comparable between runs.
# The server's cave, pond and canopy data are the benchmarks; the two
# seconds after the last are the meshes around the first view drawn.
# After a benchmark key, its line and half a second: the light blend is
# snapped (voxel_shading.update), the frames after it draw the new place.
wait_log_any 60000 benchmarks 2 and 3 ready
wait_log_any 60000 benchmark 4 ready
wait_log_any 60000 benchmarks 5, 6 and 7 ready
delay 2000
# Light through leaves is animated; F holds the clock still so the
# images are comparable between runs. See the game's README.
keypress F
wait_log 5000 wind frozen: true
keypress 1
wait_log 5000 benchmark 1 (
delay 500
screenshot ../tmp/voxel_lighting_check/1_overview.png
keypress 2
wait_log 5000 benchmark 2 (
delay 500
screenshot ../tmp/voxel_lighting_check/2_cave_mouth.png
keypress 3
wait_log 5000 benchmark 3 (
delay 500
screenshot ../tmp/voxel_lighting_check/3_inside_cave.png
keypress 4
wait_log 5000 benchmark 4 (
delay 500
screenshot ../tmp/voxel_lighting_check/4_pond.png
keypress 5
wait_log 5000 benchmark 5 (
delay 500
screenshot ../tmp/voxel_lighting_check/5_backlit_canopy.png
keypress 6
wait_log 5000 benchmark 6 (
delay 500
screenshot ../tmp/voxel_lighting_check/6_sun_behind_canopy.png
keypress 7
wait_log 5000 benchmark 7 (
delay 500
screenshot ../tmp/voxel_lighting_check/7_sun_in_sky.png

# Relight on edit: dig the shaft up out of the cave, then hang the slab over the
# mouth. All three views are shot after each, so each edit can be followed as a
# sequence of three images. View 3 is the telling one for the shaft: the part of
# the cave it opens up had no skylight at all before.
# V asks the server to verify the stored skylight; it changes nothing and
# the packets keep their order, so nothing waits for it. An edit's first
# chunk drawn again, then a second for the rest of its chunks (ten a frame).
keypress V
keypress P
wait_log 10000 drawn again
delay 1000
keypress 1
wait_log 5000 benchmark 1 (
delay 500
screenshot ../tmp/voxel_lighting_check/8_shaft_overview.png
keypress 2
wait_log 5000 benchmark 2 (
delay 500
screenshot ../tmp/voxel_lighting_check/9_shaft_cave_mouth.png
keypress 3
wait_log 5000 benchmark 3 (
delay 500
screenshot ../tmp/voxel_lighting_check/10_shaft_inside_cave.png
keypress V
keypress O
wait_log 10000 drawn again
delay 1000
keypress 1
wait_log 5000 benchmark 1 (
delay 500
screenshot ../tmp/voxel_lighting_check/11_slab_overview.png
keypress 2
wait_log 5000 benchmark 2 (
delay 500
screenshot ../tmp/voxel_lighting_check/12_slab_cave_mouth.png
keypress 3
wait_log 5000 benchmark 3 (
delay 500
screenshot ../tmp/voxel_lighting_check/13_slab_inside_cave.png
keypress V
# Dig the shaft a second time. It reaches through the cap, so this cuts the cap
# back out and the cave lights up again; the scene can be cycled like this.
keypress P
wait_log 10000 drawn again
delay 1000
keypress V
keypress 3
wait_log 5000 benchmark 3 (
delay 500
screenshot ../tmp/voxel_lighting_check/14_recycled_inside_cave.png
quit
