# Client command sequence for a visual check of the scene in eight sections.
# Run from Build/, against a server started with -u sections=8:
#   bin/buildat_server -m ../apps/voxel_lighting -u sections=8
#   bin/buildat -s localhost -w 1600x900 \
#           -c @../apps/voxel_lighting/check_multisection.txt
#
# -w fixes the window size for the run without touching the remembered one, so
# the images are the same size whatever the window was left at last time.
# See doc/client_commands.txt.
#
# The benchmark cameras are keys 1/2/3, so nothing here depends on timing a
# movement, and the three images are directly comparable between runs.
delay 8000
# Light through leaves is animated; F holds the clock still so the
# images are comparable between runs. See the game's README.
keypress F
delay 200
keypress 1
delay 500
screenshot ../tmp/multisection_lighting_check/1_overview.png
keypress 2
delay 500
screenshot ../tmp/multisection_lighting_check/2_cave_mouth.png
keypress 3
delay 500
screenshot ../tmp/multisection_lighting_check/3_inside_cave.png

# Relight on edit: dig the shaft up out of the cave, then hang the slab over the
# mouth. All three views are shot after each, so each edit can be followed as a
# sequence of three images. View 3 is the telling one for the shaft: the part of
# the cave it opens up had no skylight at all before.
keypress V
delay 1500
keypress P
delay 1500
keypress 1
delay 500
screenshot ../tmp/multisection_lighting_check/4_shaft_overview.png
keypress 2
delay 500
screenshot ../tmp/multisection_lighting_check/5_shaft_cave_mouth.png
keypress 3
delay 500
screenshot ../tmp/multisection_lighting_check/6_shaft_inside_cave.png
keypress V
delay 1500
keypress O
delay 1500
keypress 1
delay 500
screenshot ../tmp/multisection_lighting_check/7_slab_overview.png
keypress 2
delay 500
screenshot ../tmp/multisection_lighting_check/8_slab_cave_mouth.png
keypress 3
delay 500
screenshot ../tmp/multisection_lighting_check/9_slab_inside_cave.png
keypress V
delay 1500
# Dig the shaft a second time. It reaches through the cap, so this cuts the cap
# back out and the cave lights up again; the scene can be cycled like this.
keypress P
delay 1500
keypress V
delay 1500
keypress 3
delay 500
screenshot ../tmp/multisection_lighting_check/10_recycled_inside_cave.png
