# Client command sequence for a visual check. Run from Build/:
#   bin/buildat -s localhost -w 1600x900 \
#           -c @../apps/sample_falling_blocks/check.txt
#
# -w fixes the window size for the run without touching the remembered one, so
# the images are the same size whatever the window was left at last time.
# See doc/client_commands.txt.
#
# The camera does not move, so the only difference between the images is how
# far the pile has got. The server drops a piece every 0.7 s and reuses the
# oldest one once there are 24 of them, so the pile stops growing by itself.
delay 8000
screenshot ../tmp/voxel_physics_check/1_first_pieces.png
delay 10000
screenshot ../tmp/voxel_physics_check/2_pile.png
delay 10000
screenshot ../tmp/voxel_physics_check/3_full.png
# R clears the pieces; the next drop starts the pile again from nothing
keypress R
delay 1000
screenshot ../tmp/voxel_physics_check/4_cleared.png
